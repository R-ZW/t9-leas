===== INÍCIO DO ARQUIVO: README.md (119 linhas) =====
  1| # foremost-ng
  2| 
  3| `foremost-ng` is a Linux-based file recovery tool that extracts files from disk images or devices by scanning for configurable file headers and footers. It supports common forensic formats (e.g., `.dd`, Safeback, Encase) and can operate on both image files and live devices.
  4| > ⚡ This fork aims to modernize the codebase, improve terminal output with ANSI styling, and introduce new features for forensic analysis and data recovery.
  5| 
  6| <img src="screenshots/foremost-ng-01.png"/>
  7| <img src="screenshots/foremost-ng-02.png"/>
  8| 
  9| ---
 10| 
 11| ## Features
 12| 
 13| - 🧩 Recover files based on headers and footers.
 14| - ⚙️ Configurable file signatures via a plain-text configuration file.
 15| - 🪛 Supports header-only or header–footer scans.
 16| - 🧱 Works on forensic images or live drives.
 17| - 🦠 VirusTotal lookup by file hash (requires a VT API key).
 18| 
 19| ---
 20| 
 21| ## Installation
 22| 
 23| Clone or download the `foremost-ng` archive, then build and install according to your platform:
 24| 
 25| ```bash
 26| # Common steps
 27| tar xzvf foremost-ng-<version>.tar.gz
 28| cd foremost-ng-<version>/src
 29| ```
 30| 
 31| ### Linux
 32| ```bash
 33| make
 34| sudo make install
 35| ```
 36| 
 37| ### BSD
 38| 
 39| Make sure ```curl``` is installed:
 40| ```bash
 41| pkg_add curl # OpenBSD
 42| pkg install curl # FreeBSD
 43| ```
 44| 
 45| ```bash
 46| make unix
 47| sudo gmake install
 48| ```
 49| 
 50| ### macOS
 51| 
 52| Install OpenSSL:
 53| ```bash
 54| brew install openssl@3
 55| ```
 56| 
 57| ```bash
 58| make mac
 59| sudo make macinstall
 60| ```
 61| 
 62| ### Cross compiling for Windows
 63| 
 64| Install MinGW in your Linux distribution and cross-compile:
 65| ```bash
 66| make cross
 67| ```
 68| **Required DLLs can be found in the [windows_dlls](windows_dlls) directory of this repository.**
 69| 
 70| > **Note:** On systems with glibc < 2.2.0, you may see harmless warnings about `ftello` and `fseeko`. These can be safely ignored.
 71| 
 72| ---
 73| 
 74| ## Usage
 75| 
 76| Run `foremost-ng` with the appropriate command-line options. For full details, refer to the manual page:
 77| 
 78| ```bash
 79| man foremost-ng
 80| ```
 81| 
 82| Basic syntax:
 83| ```bash
 84| foremost-ng [options] [image_or_device]
 85| ```
 86| 
 87| ---
 88| 
 89| ## VirusTotal API
 90| 
 91| `foremost-ng` supports automatic file reputation checks using the VirusTotal API via the ```-x``` command-line option. This allows you to analyze recovered files for potential threats by submitting their hashes to VirusTotal.
 92| 
 93| To enable this feature, follow these steps:
 94| 
 95| 1. [Create a free VirusTotal account](https://www.virustotal.com/gui/join-us).
 96| 2. Obtain your personal **API key**.
 97| 3. Set the key as an environment variable named `VT_API_KEY`.
 98| 
 99| ```bash
100| # Unix-like systems
101| export VT_API_KEY=yourkey
102| 
103| # Windows CMD
104| set VT_API_KEY=yourkey
105| 
106| # Windows PowerShell
107| $env:VT_API_KEY="yourkey"
108| ```
109| 
110| ---

112| ## Uninstallation
113| 
114| To remove `foremost-ng` from your system:
115|

117| cd foremost-ng-<version>/src
118| sudo make uninstall

 59| /**
 60|  * Perform modified Boyer-Moore search with wildcards and custom start
 61|  */
 62| unsigned char *bm_search_skipn(
 63|     unsigned char *needle, size_t needle_len,
 64|     unsigned char *haystack, size_t haystack_len,
 65|     size_t table[UCHAR_MAX + 1],
 66|     int casesensitive, int searchtype, int start_pos
 67| ) {
 68|     size_t shift, pos = start_pos;
 69|     unsigned char *here;
 70| 
 71|     if (needle_len == 0)
 72|         return haystack;
 73| 
 74|     if (searchtype == SEARCHTYPE_FORWARD || searchtype == SEARCHTYPE_FORWARD_NEXT) {
 75|         // Forward search using bad-character shifts
 76|         while (pos < haystack_len) {
 77|             // Jump ahead by shift value
 78|             while (pos < haystack_len &&
 79|                    (shift = table[(unsigned char)haystack[pos]]) > 0) {
 80|                 pos += shift;
 81|             }
 82|             if (shift == 0) {
 83|                 here = &haystack[pos - needle_len + 1];
 84|                 // Compare allowing wildcards
 85|                 if (memwildcardcmp(needle, here, needle_len, casesensitive) == 0) {
 86|                     return here;
 87|                 }
 88|                 pos++;
 89|             }
 90|         }
 91|         return NULL;
 92|     }
 93|     else if (searchtype == SEARCHTYPE_REVERSE) {
 94|         // Reverse search scanning from buffer end
 95|         while (pos < haystack_len) {
 96|             while (pos < haystack_len &&
 97|                    (shift = table[(unsigned char)haystack[haystack_len - pos - 1]]) > 0) {
 98|                 pos += shift;
 99|             }
100|             if (shift == 0) {
101|                 here = &haystack[haystack_len - pos - 1];
102|                 if (memwildcardcmp(needle, here, needle_len, casesensitive) == 0) {
103|                     return here;
104|                 }
105|                 pos++;
106|             }
107|         }
108|         return NULL;
109|     }
110|     return NULL;

198| /**

200|  */
201| int ind_block(unsigned char *foundat, uint64_t buflen, int bs) {
202|     unsigned char *temp = foundat;
203|     int jump = 12 * bs;                         /* Offset to pointers */
204|     unsigned int block, block2, dif;
205|     int i = 0;
206|     unsigned int numbers = (bs / 4) - 1;
207| 
208|     // Ensure buffer contains at least two pointers
209|     if (buflen < (uint64_t)jump + 16)
210|         return FALSE;
211| 
212|     // Check sequential pointer increments
213|     while (i < (int)numbers) {
214|         block = htoi(&temp[jump + i*4], FOREMOST_LITTLE_ENDIAN);
215|         if (block == 0) break;
216|         i++;
217|         block2 = htoi(&temp[jump + i*4], FOREMOST_LITTLE_ENDIAN);
218|         if (block2 == 0) break;
219|         dif = block2 - block;
220|         if (dif != 1) return FALSE;
221|     }
222|     if (i == 0) return FALSE;
223| 
224|     // Verify remaining pointers are zero
225|     for (i = i + 1; i < (int)numbers; i++) {
226|         if (htoi(&temp[jump + i*4], FOREMOST_LITTLE_ENDIAN) != 0)
227|             return FALSE;
228|     }

 526| /**
 527|  * Check basic OLE header fields (reserved, sector shifts, FAT counts)
 528|  * to decide whether an OLE container looks valid.
 529|  */
 530| int valid_ole_header(struct OLE_HDR *h) {
 531| 
 532|     if (htos((unsigned char *) &h->reserved, FOREMOST_LITTLE_ENDIAN) != 0 ||
 533|         htoi((unsigned char *) &h->reserved1, FOREMOST_LITTLE_ENDIAN) != 0 ||
 534|         htoi((unsigned char *) &h->reserved2, FOREMOST_LITTLE_ENDIAN) != 0) {
 535|         return FALSE;
 536|     }
 537| 
 538|     // The minimum sector shift is usually 2^6(64) and the uSectorShift is 2^9(512))
 539|     if (htos((unsigned char *) &h->uMiniSectorShift, FOREMOST_LITTLE_ENDIAN) != 6 ||
 540|         htos((unsigned char *) &h->uSectorShift, FOREMOST_LITTLE_ENDIAN) != 9 ||
 541|         htoi((unsigned char *) &h->dir_flag, FOREMOST_LITTLE_ENDIAN) < 0) {
 542|         return FALSE;
 543|     }
 544| 
 545|     // Sanity Checking
 546|     if (htoi((unsigned char *) &h->num_FAT_blocks, FOREMOST_LITTLE_ENDIAN) <= 0 ||
 547|         htoi((unsigned char *) &h->num_FAT_blocks, FOREMOST_LITTLE_ENDIAN) > 100) {
 548|         return FALSE;
 549|     }
 550| 
 551|     if (htoi((unsigned char *) &h->num_extra_FAT_blocks, FOREMOST_LITTLE_ENDIAN) < 0 ||
 552|         htoi((unsigned char *) &h->num_extra_FAT_blocks, FOREMOST_LITTLE_ENDIAN) > 100) {
 553|         return FALSE;
 554|     }
 555| 
 556|     return TRUE;
 557| }

 61|         md = EVP_sha1();
 62|     } else if (strcmp(algo, "sha256") == 0) {
 63|         md = EVP_sha256();

 66|         fclose(file);

 78|         fprintf(stderr, "EVP_DigestInit_ex failed\n");

 81|         exit(EXIT_FAILURE);
 82|     }

 86|     while ((bytesRead = fread(buf, 1, sizeof(buf), file)) > 0) {

 89|             fclose(file);
